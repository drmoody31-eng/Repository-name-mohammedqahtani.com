// mq-scan: reads a product photo for mohammedqahtani.com/product-check/
// Secret required: ANTHROPIC_API_KEY. Optional variable: MODEL. Optional KV binding STATS: private daily counts.
const CATS = { hair: 'shampoo|conditioner|mask|leavein|oil|serum|styling|scalp|other', skin: 'cleanser|moisturizer|sunscreen|serum|toner|eye|mask|other' };
const TEX = { hair: '[] (leave empty)', skin: '"light" (gel, fluid, lotion or oil-free), "rich" (thick cream, balm or ointment), "matte", "foam" (foaming cleanser), "gentle" (cream, milk or micellar cleanser), "mineral" (mineral-only sunscreen)' };
const GUIDE = { hair: '"ss" (sodium/ammonium lauryl or laureth sulfate), "mild" (shampoo without those sulfates), "ad" (ketoconazole, zinc pyrithione, piroctone olamine, selenium sulfide, climbazole), "sal" (salicylic acid), "sil" (silicones), "heavy" (coconut, castor, olive oil, shea butter, mineral oil), "light" (argan, jojoba, squalane, grapeseed), "protein" (hydrolyzed proteins, keratin, amino acids), "bond" (bis-aminopropyl diglycol dimaleate), "hum" (glycerin, panthenol, aloe), "alc" (alcohol denat), "frag" (parfum/fragrance), "fragfree" (no fragrance)', skin: '"gentle" (non-foaming cream or gentle cleanser), "foam" (strong foaming cleanser), "sa" (salicylic acid), "bpo" (benzoyl peroxide), "aha" (glycolic, lactic, mandelic acid), "ret" (retinol, retinal, adapalene), "vitc" (vitamin C), "nia" (niacinamide), "aza" (azelaic acid), "ha" (hyaluronic acid, glycerin, panthenol), "cer" (ceramides), "rich" (thick occlusive cream: petrolatum, shea, mineral oil), "light" (gel, fluid or oil-free texture), "comedo" (coconut oil, cocoa butter, isopropyl myristate), "matte" (mattifying), "alc" (alcohol denat), "frag" (parfum/fragrance), "fragfree" (no fragrance), "spf" (sunscreen), "mineral" (zinc oxide or titanium dioxide filters only)' };
const CATCREAM = { hair: '"leavein", a scalp tonic or lotion is "scalp"', skin: '"moisturizer", a correcting or active treatment is "serum"' };
function promptFor(mode) {
  return 'You identify a cosmetic product from a photo for a product-suitability tool. Reply with ONE JSON object and nothing else:\n' +
    '{"readable":true|false,"domain":"hair|skin|other","brand":"brand in English","name":"product name in English as printed, without the brand","category":"' + CATS[mode] + '","texture":[...],"traits":[...],"confidence":"high|medium|low","purpose":"treat|care","drug":"medicine actives in this product if any (for example minoxidil, finasteride, dutasteride, tretinoin, hydroquinone, a corticosteroid, an antibiotic, an antifungal), else empty","ingredients":"the ingredient (INCI) list copied from the photo if it is visible, else empty","known_ingredients":"see below"}\n' +
    'domain: "hair" for hair or scalp products, "skin" for face or body skincare, "other" for anything else. This request is for ' + mode + ' products; if the product belongs to the other domain still fill brand, name and domain.\n' +
    'purpose: "treat" when the product is sold to treat a hair, scalp or skin condition (hair loss, regrowth, thinning, alopecia, dandruff, psoriasis and similar); "care" when it is sold to improve the look or quality of hair or skin (shine, softness, frizz, hydration, repair, volume, styling, cleansing).\n' +
    'category: choose the closest one; a treatment or correcting cream is ' + CATCREAM[mode] + '. Use "other" only when none could fit.\n' +
    'known_ingredients: only when the ingredient list is NOT visible in the photo. If you recognise this exact product and reliably know its published ingredient list, write that full INCI list in order, comma separated, and set confidence to "high" (you are sure of the list) or "medium" (sure of the main ingredients, less sure of the order or the tail). ' +
    "If you do not know the full list but you do know the main ingredients of this exact product, write those main ingredients (at least 8, most important first) and set confidence " + '"medium". You do not need to be certain: if you are about 80% sure of the main ingredients of this exact product, give them with confidence "medium". Only if you know nothing about its formula, leave it empty and set confidence "low". Do not substitute a different product from the same brand.\n' +
    'texture: any of ' + TEX[mode] + ' that clearly apply to this product.\n' +
    'traits: used for an approximate rating when the formula is not known. From this list, include every trait that applies to this product according to the claims printed on the pack and what you know about this product line: ' + GUIDE[mode] + '.' + (mode === 'hair' ? ' For a shampoo always include either "ss" or "mild". For a hair oil always include either "light" or "heavy", and for a pure single oil write that oil as known_ingredients with confidence "high".' : '') + ' Whenever readable is true and the product belongs to this domain, traits must NOT be empty, even when the photo shows only the front of the pack: if you are not sure of the formula, give the traits that follow from the product name, the claims printed on the pack, its category and what this product line is known for (for example a repair, bond, peptide or keratin product is "protein" or "bond"; a conditioner, mask or leave-in is "hum"; a clarifying or anti-dandruff shampoo is "ss" or "ad"; a hydrating serum or cream is "ha"; a sunscreen is "spf").\n' +
    'If nothing is legible or it is not a cosmetic product, set readable false. Never invent a brand or a name.';
}
const ORIGINS = ['https://mohammedqahtani.com', 'https://www.mohammedqahtani.com'];
const PER_IP_PER_DAY = 30;
const hits = new Map();
function cors(origin) {
  return { 'Access-Control-Allow-Origin': ORIGINS.includes(origin) ? origin : ORIGINS[0], 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'content-type', 'Access-Control-Max-Age': '86400', 'Vary': 'Origin' };
}
function reply(body, status, origin) {
  return new Response(JSON.stringify(body), { status: status, headers: Object.assign({ 'content-type': 'application/json; charset=utf-8', 'cache-control': 'no-store' }, cors(origin)) });
}
const EVENTS = ['quiz', 'name', 'paste', 'order', 'card'];
async function count(env, mode, ip, ev, from) {
  if (!env.STATS) return;
  try {
    const day = new Date(Date.now() + 10800000).toISOString().slice(0, 10);
    const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(day + '|' + ip));
    const vk = 'v:' + day + ':' + Array.from(new Uint8Array(buf).slice(0, 8)).map(function (b) { return b.toString(16).padStart(2, '0'); }).join('');
    const seen = await env.STATS.get(vk);
    if (!seen) await env.STATS.put(vk, '1', { expirationTtl: 172800 });
    const dk = 'd:' + day;
    const cur = JSON.parse((await env.STATS.get(dk)) || '{}');
    if (ev) cur[ev] = (cur[ev] || 0) + 1;
    else { cur.scans = (cur.scans || 0) + 1; cur[mode] = (cur[mode] || 0) + 1; }
    if (!seen) cur.visitors = (cur.visitors || 0) + 1;
    /* where the visitor came from (instagram, tiktok, snapchat, a tagged link): devices, quizzes, order clicks and scans per source */
    const f = String(from || '').toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 12);
    if (f) {
      if (!seen) cur['f_' + f + '_dev'] = (cur['f_' + f + '_dev'] || 0) + 1;
      const e = ev || 'scan';
      if (e === 'quiz' || e === 'order' || e === 'scan') cur['f_' + f + '_' + e] = (cur['f_' + f + '_' + e] || 0) + 1;
    }
    await env.STATS.put(dk, JSON.stringify(cur));
  } catch (e) {}
}

const SRC = ['db', 'inci', 'known', 'approx', 'fail', 'unread', 'med'];
const IDPROMPT = 'Read the cosmetic product in this photo. Reply with ONE JSON object and nothing else: {"readable":true|false,"brand":"brand in English","name":"product name in English as printed, without the brand"}. If nothing is legible set readable false. Never invent a brand or a name.';
function pkey(mode, name) {
  const t = String(name || '').toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/[^a-z0-9]+/g, ' ').trim().split(' ').filter(Boolean);
  const u = Array.from(new Set(t)).sort().join('-').slice(0, 150);
  return u.length > 3 ? 'p:' + mode + ':' + u : '';
}
function parse(t) { const m = String(t || '').match(/\{[\s\S]*\}/); if (!m) return null; try { return JSON.parse(m[0]); } catch (e) { return null; } }
function fullName(j) { return ((j && j.brand || '') + ' ' + (j && j.name || '')).trim(); }
function listed(x) { return typeof x === 'string' && x.split(',').length >= 6; }
function day() { return new Date(Date.now() + 10800000).toISOString().slice(0, 10); }
async function ask(env, image, prompt, max, signal) {
  const r = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST', signal: signal,
    headers: { 'x-api-key': env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01', 'content-type': 'application/json' },
    body: JSON.stringify({ model: env.MODEL || 'claude-sonnet-5-5', max_tokens: max, messages: [{ role: 'user', content: [{ type: 'image', source: { type: 'base64', media_type: 'image/jpeg', data: image } }, { type: 'text', text: prompt }] }] })
  });
  if (!r.ok) throw new Error('upstream ' + r.status);
  const j = await r.json();
  return (j.content || []).filter(function (c) { return c.type === 'text'; }).map(function (c) { return c.text; }).join('');
}
/* products seen so far: p:<mode>:<name> keeps the last full reading, so the same product is answered at once next time */
async function recGet(env, k) { if (!env.STATS || !k) return null; try { return JSON.parse((await env.STATS.get(k)) || 'null'); } catch (e) { return null; } }
async function recPut(env, k, rec) {
  if (!env.STATS || !k) return;
  try { await env.STATS.put(k, JSON.stringify(rec), { metadata: { n: String(rec.name || '').slice(0, 120), c: rec.c || 0, s: rec.src || '', m: rec.mode, g: rec.good ? 1 : 0, t: rec.t || '' } }); } catch (e) {}
}
async function remember(env, mode, j, fromCache) {
  const k = pkey(mode, fullName(j)); if (!k) return;
  const old = await recGet(env, k);
  const rec = old || { mode: mode, name: fullName(j), c: 0, src: '' };
  rec.c = (rec.c || 0) + 1; rec.t = day();
  if (!fromCache) {
    const hasInci = listed(j.ingredients), hasKnown = listed(j.known_ingredients) && j.confidence !== 'low';
    const oldInci = rec.j && rec.j.confidence === 'high' && rec.fromInci;
    if (hasInci) { rec.j = Object.assign({}, j, { ingredients: '', known_ingredients: j.ingredients, confidence: 'high' }); rec.good = true; rec.fromInci = true; }
    else if (!oldInci && (hasKnown || !rec.j)) { rec.j = j; rec.good = !!hasKnown; }
  }
  await recPut(env, k, rec);
}
async function logResult(env, mode, name, src) {
  const k = pkey(mode, name); if (!k) return;
  const rec = (await recGet(env, k)) || { mode: mode, name: String(name).slice(0, 120), c: 1, t: day() };
  if (rec.src === src) return;
  rec.src = src; await recPut(env, k, rec);
}
export default {
  async fetch(request, env, ctx) {
    const origin = request.headers.get('Origin') || '';
    if (request.method === 'OPTIONS') return new Response(null, { status: 204, headers: cors(origin) });
    if (request.method === 'GET') return reply({ ready: !!env.ANTHROPIC_API_KEY }, 200, origin);
    if (request.method !== 'POST') return reply({ error: 'method' }, 405, origin);
    if (!ORIGINS.includes(origin)) return reply({ error: 'origin' }, 403, origin);
    if (!env.ANTHROPIC_API_KEY) return reply({ error: 'not_configured' }, 503, origin);
    const ip = request.headers.get('CF-Connecting-IP') || 'x';
    const key = new Date().toISOString().slice(0, 10) + '|' + ip;
    if (hits.size > 5000) hits.clear();
    const n = (hits.get(key) || 0) + 1;
    hits.set(key, n);
    if (n > PER_IP_PER_DAY) return reply({ error: 'limit' }, 429, origin);
    let body;
    try { body = await request.json(); } catch (e) { return reply({ error: 'json' }, 400, origin); }
    const mode = body && body.mode, image = body && body.image;
    if (body && EVENTS.includes(body.event) && CATS[mode]) { hits.set(key, n - 1); ctx.waitUntil(count(env, mode, ip, body.event, body.from)); return reply({ ok: true }, 200, origin); }
    if (body && SRC.includes(body.log) && CATS[mode]) { hits.set(key, n - 1); if (body.log === 'unread') ctx.waitUntil(count(env, mode, ip, 'unread')); else if (typeof body.name === 'string' && body.name.length < 200) ctx.waitUntil(logResult(env, mode, body.name, body.log)); return reply({ ok: true }, 200, origin); }
    if (!CATS[mode] || typeof image !== 'string' || image.length < 200 || image.length > 2800000 || !/^[A-Za-z0-9+/=]+$/.test(image)) return reply({ error: 'input' }, 400, origin);
    const ac = new AbortController();
    const full = ask(env, image, promptFor(mode), 3000, ac.signal).then(function (t) { return { text: t }; }, function () { return { fail: true }; });
    /* a short reading of the name runs beside the full one; a product read before is answered from memory */
    const quick = ask(env, image, IDPROMPT, 200).then(async function (t) {
      const q = parse(t); if (!q || !q.readable) return null;
      const rec = await recGet(env, pkey(mode, fullName(q)));
      return rec && rec.j && (rec.good || rec.src === 'db') ? rec : null;
    }).catch(function () { return null; });
    const first = await Promise.race([quick.then(function (rec) { return rec ? { rec: rec } : full; }), full]);
    if (first.rec) {
      try { ac.abort(); } catch (e) {}
      ctx.waitUntil(count(env, mode, ip, '', body.from)); ctx.waitUntil(remember(env, mode, first.rec.j, true));
      return reply({ text: JSON.stringify(first.rec.j), cached: true }, 200, origin);
    }
    if (first.fail) return reply({ error: 'upstream' }, 502, origin);
    ctx.waitUntil(count(env, mode, ip, '', body.from));
    const pj = parse(first.text);
    if (pj && pj.readable && pj.domain !== (mode === 'hair' ? 'skin' : 'hair')) ctx.waitUntil(remember(env, mode, pj, false));
    return reply({ text: first.text }, 200, origin);
  }
};
